package view;

import javax.swing.*;
import java.awt.*;
import java.awt.event.ItemEvent;
import java.sql.*;

import database.DBConnection;

public class StockForm extends JFrame {

    JComboBox<String> productBox;
    JTextField quantityField;
    JTextField priceField;
    JButton addButton;
    JButton removeButton;
    JButton updatePriceButton;

    private Dashboard previousDashboard;

    private final Color DARK_BLUE = new Color(31, 52, 80);
    private final Color BACKGROUND = new Color(240, 245, 250);
    private final Color GREEN = new Color(35, 150, 105);
    private final Color RED = new Color(190, 65, 65);
    private final Color BLUE = new Color(40, 120, 220);
    private final Color HEADER_BLUE = new Color(32, 57, 92);

    public StockForm() {
        this(null);
    }

        priceField.setBounds(230, 250, 220, 35);

    private void loadProducts() {

        String sql = "SELECT ProductName FROM Products ORDER BY ProductName";

        try {
            Connection con = DBConnection.getConnection();
            PreparedStatement pst = con.prepareStatement(sql);
            ResultSet rs = pst.executeQuery();

            productBox.removeAllItems();

            while (rs.next()) {
                productBox.addItem(rs.getString("ProductName"));
            }

            rs.close();
            pst.close();
            con.close();

            loadSelectedProductPrice();

        } catch (Exception e) {
            JOptionPane.showMessageDialog(
                    this,
                    "Unable to load products.\n\n" + e.getMessage(),
                    "Database Error",
                    JOptionPane.ERROR_MESSAGE
            );
        }
    }

    private void loadSelectedProductPrice() {

        if (productBox.getSelectedItem() == null) {
            priceField.setText("");
            return;
        }

        String product = productBox.getSelectedItem().toString();

        String sql = "SELECT Price FROM Products WHERE ProductName = ?";

        try {
            Connection con = DBConnection.getConnection();
            PreparedStatement pst = con.prepareStatement(sql);
            pst.setString(1, product);

            ResultSet rs = pst.executeQuery();

            if (rs.next()) {
                double price = rs.getDouble("Price");
                priceField.setText(String.format("%.2f", price));
            }

            rs.close();
            pst.close();
            con.close();

        } catch (Exception e) {
            System.out.println("Error loading price: " + e.getMessage());
        }
    }

    private void updateStock(boolean add) {

        try {
            if (productBox.getSelectedItem() == null) {
                JOptionPane.showMessageDialog(
                        this,
                        "Please select a product.",
                        "Input Error",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }

            String product = productBox.getSelectedItem().toString();
            String quantityText = quantityField.getText().trim();

            if (quantityText.isEmpty()) {
                JOptionPane.showMessageDialog(
                        this,
                        "Please enter a quantity.",
                        "Input Error",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }

            int quantity = Integer.parseInt(quantityText);

            if (quantity <= 0) {
                JOptionPane.showMessageDialog(
                        this,
                        "Quantity must be greater than 0.",
                        "Input Error",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }

            Connection con = DBConnection.getConnection();

            if (add) {
                String sql = "UPDATE Products SET Quantity = Quantity + ? WHERE ProductName = ?";

                PreparedStatement pst = con.prepareStatement(sql);
                pst.setInt(1, quantity);
                pst.setString(2, product);
                pst.executeUpdate();
                pst.close();

                JOptionPane.showMessageDialog(
                        this,
                        "Stock Added Successfully!",
                        "Success",
                        JOptionPane.INFORMATION_MESSAGE
                );

            } else {
                String checkSql = "SELECT Quantity FROM Products WHERE ProductName = ?";

                PreparedStatement checkPst = con.prepareStatement(checkSql);
                checkPst.setString(1, product);

                ResultSet rs = checkPst.executeQuery();

                if (rs.next()) {
                    int currentQuantity = rs.getInt("Quantity");

                    if (quantity > currentQuantity) {
                        rs.close();
                        checkPst.close();
                        con.close();

                        JOptionPane.showMessageDialog(
                                this,
                                "Cannot remove " + quantity + " items.\n\nAvailable stock: " + currentQuantity,
                                "Insufficient Stock",
                                JOptionPane.WARNING_MESSAGE
                        );
                        return;
                    }
                }

                rs.close();
                checkPst.close();

                String sql = "UPDATE Products SET Quantity = Quantity - ? WHERE ProductName = ?";

                PreparedStatement pst = con.prepareStatement(sql);
                pst.setInt(1, quantity);
                pst.setString(2, product);
                pst.executeUpdate();
                pst.close();

                JOptionPane.showMessageDialog(
                        this,
                        "Stock Removed Successfully!",
                        "Success",
                        JOptionPane.INFORMATION_MESSAGE
                );
            }

            con.close();
            quantityField.setText("");

        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(
                    this,
                    "Please enter a valid number.",
                    "Input Error",
                    JOptionPane.WARNING_MESSAGE
            );
        } catch (Exception e) {
            JOptionPane.showMessageDialog(
                    this,
                    "Error: " + e.getMessage(),
                    "Database Error",
                    JOptionPane.ERROR_MESSAGE
            );
        }
    }

    private void updatePrice() {

        try {
            if (productBox.getSelectedItem() == null) {
                JOptionPane.showMessageDialog(
                        this,
                        "Please select a product.",
                        "Input Error",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }

            String product = productBox.getSelectedItem().toString();
            String priceText = priceField.getText().trim();

            if (priceText.isEmpty()) {
                JOptionPane.showMessageDialog(
                        this,
                        "Please enter a new price.",
                        "Input Error",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }

            double newPrice;

            try {
                newPrice = Double.parseDouble(priceText);
            } catch (NumberFormatException e) {
                JOptionPane.showMessageDialog(
                        this,
                        "Please enter a valid price.",
                        "Input Error",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }

            if (newPrice < 0) {
                JOptionPane.showMessageDialog(
                        this,
                        "Price cannot be negative.",
                        "Invalid Price",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }

            Connection con = DBConnection.getConnection();

            String sql = "UPDATE Products SET Price = ? WHERE ProductName = ?";

            PreparedStatement pst = con.prepareStatement(sql);
            pst.setDouble(1, newPrice);
            pst.setString(2, product);

            int rows = pst.executeUpdate();

            pst.close();
            con.close();

            if (rows > 0) {
                JOptionPane.showMessageDialog(
                        this,
                        "Price Updated Successfully!\n\n"
                                + "Product: " + product + "\n"
                                + "New Price: R" + String.format("%.2f", newPrice),
                        "Success",
                        JOptionPane.INFORMATION_MESSAGE
                );

                loadSelectedProductPrice();
            }

        } catch (Exception e) {
            JOptionPane.showMessageDialog(
                    this,
                    "Error updating price.\n\n" + e.getMessage(),
                    "Database Error",
                    JOptionPane.ERROR_MESSAGE
            );
        }
    }

    private static class RoundedButton extends JButton {

        public RoundedButton(String text) {

            super(text);
            setOpaque(false);
            setContentAreaFilled(false);
            setBorderPainted(false);
            setFocusPainted(false);
            setCursor(new Cursor(Cursor.HAND_CURSOR));
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 = (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            g2.setColor(getBackground());
            g2.fillRoundRect(0, 0, getWidth(), getHeight(), 18, 18);

            g2.setFont(getFont());
            g2.setColor(getForeground());

            FontMetrics fm = g2.getFontMetrics();

            int x = (getWidth() - fm.stringWidth(getText())) / 2;
            int y = (getHeight() - fm.getHeight()) / 2 + fm.getAscent();

            g2.drawString(getText(), x, y);
            g2.dispose();
        }

        @Override
        protected void paintBorder(Graphics g) {
        }
    }
