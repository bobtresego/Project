    private void loadGraphData() {

        List<ProductInfo> products = new ArrayList<>();

        String query = "SELECT ProductID, ProductName, Category, Quantity, Price FROM Products ORDER BY ProductID";

        Connection con = null;

        try {
            con = DBConnection.getConnection();
            if (con == null) {
                System.out.println("Database connection is null.");
                return;
            }

            try (
                    PreparedStatement pst = con.prepareStatement(query);
                    ResultSet rs = pst.executeQuery()
            ) {
                while (rs.next()) {
                    ProductInfo product = new ProductInfo(
                            rs.getInt("ProductID"),
                            rs.getString("ProductName"),
                            rs.getString("Category"),
                            rs.getInt("Quantity"),
                            rs.getDouble("Price")
                    );
                    products.add(product);
                }
            }

        } catch (Exception e) {
            System.out.println("Error loading report data: " + e.getMessage());
        } finally {
            if (con != null) {
                try { con.close(); } catch (SQLException ignored) {}
            }
        }

        stockChart.setData(products);
        priceChart.setData(products);
    }

    private static void showProductInformation(ProductInfo product) {

        if (product == null) return;

        String category = product.category;
        if (category == null || category.trim().isEmpty()) {
            category = "Not specified";
        }

        String stockWarning = "";
        if (product.quantity < LOW_STOCK_THRESHOLD) {
            stockWarning =
                    "<p style='color:#c83737;'><b>⚠ LOW STOCK WARNING</b><br>"
                            + "This product is below the minimum threshold of "
                            + LOW_STOCK_THRESHOLD + " units.</p>";
        }

        String message =
                "<html>" +
                "<div style='width:330px;'>" +
                "<h2 style='color:#284678;'>" +
                "Product Information" +
                "</h2>" +
                "<hr>" +
                "<b>Product ID:</b> " + product.id +
                "<br><br>" +
                "<b>Product Name:</b> " + product.name +
                "<br><br>" +
                "<b>Category:</b> " + category +
                "<br><br>" +
                "<b>Quantity in Stock:</b> " + product.quantity +
                "<br><br>" +
                "<b>Price:</b> R" +
                String.format("%.2f", product.price) +
                stockWarning +
                "</div></html>";

        JOptionPane.showMessageDialog(
                null,
                message,
                "Product Details",
                JOptionPane.INFORMATION_MESSAGE
        );
    }

    private static class StockBarChartPanel extends JPanel {

        private List<ProductInfo> products = new ArrayList<>();

        public StockBarChartPanel() {

            setOpaque(false);
            setCursor(new Cursor(Cursor.HAND_CURSOR));

            addMouseListener(new MouseAdapter() {
                @Override
                public void mouseClicked(MouseEvent e) {
                    showClickedProduct(e.getX(), e.getY());
                }
            });
        }

        public void setData(List<ProductInfo> products) {
            this.products = new ArrayList<>(products);
            repaint();
        }

        private void showClickedProduct(int mouseX, int mouseY) {

            if (products.isEmpty()) return;

            int width = getWidth();
            int height = getHeight();

            int left = 55;
            int right = 20;
            int top = 10;
            int bottom = 45;

            int graphWidth = width - left - right;
            int graphHeight = height - top - bottom;

            int maxValue = 1;
            for (ProductInfo product : products) {
                if (product.quantity > maxValue) {
                    maxValue = product.quantity;
                }
            }

            int count = products.size();
            int space = graphWidth / Math.max(count, 1);
            int barWidth = Math.max(15, Math.min(45, space - 12));

            for (int i = 0; i < count; i++) {

                ProductInfo product = products.get(i);
                int value = product.quantity;

                int barHeight;
                if (maxValue <= 0) {
                    barHeight = 0;
                } else {
                    barHeight = (int) ((double) value / maxValue * graphHeight);
                }

                int x = left + i * space + (space - barWidth) / 2;
                int y = top + graphHeight - barHeight;

                Rectangle bar = new Rectangle(
                        x, y, barWidth, Math.max(barHeight, 1)
                );

                Rectangle clickableArea = new Rectangle(
                        x - 10,
                        Math.max(0, y - 15),
                        barWidth + 20,
                        graphHeight + 40
                );

                if (bar.contains(mouseX, mouseY)
                        || clickableArea.contains(mouseX, mouseY)) {
                    showProductInformation(product);
                    return;
                }
            }
        }

        @Override
        protected void paintComponent(Graphics g) {

            super.paintComponent(g);
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int width = getWidth();
            int height = getHeight();

            int left = 55;
            int right = 20;
            int top = 10;
            int bottom = 45;

            int graphWidth = width - left - right;
            int graphHeight = height - top - bottom;

            if (products.isEmpty()) {
                drawNoData(g2, width, height);
                g2.dispose();
                return;
            }

            int maxValue = 1;
            for (ProductInfo product : products) {
                if (product.quantity > maxValue) {
                    maxValue = product.quantity;
                }
            }

            g2.setFont(new Font("Arial", Font.PLAIN, 10));

            for (int i = 0; i <= 5; i++) {

                int y = top + graphHeight - i * graphHeight / 5;
                int value = i * maxValue / 5;

                g2.setColor(GRID_COLOR);
                g2.drawLine(left, y, width - right, y);

                g2.setColor(TEXT_LIGHT);
                g2.drawString(String.valueOf(value), 10, y + 4);
            }

            g2.setColor(AXIS_COLOR);
            g2.drawLine(left, top, left, top + graphHeight);
            g2.drawLine(left, top + graphHeight, width - right, top + graphHeight);

            int count = products.size();
            int space = graphWidth / Math.max(count, 1);
            int barWidth = Math.max(15, Math.min(45, space - 12));

            for (int i = 0; i < count; i++) {

                ProductInfo product = products.get(i);
                int value = product.quantity;

                int barHeight;
                if (maxValue <= 0) {
                    barHeight = 0;
                } else {
                    barHeight = (int) ((double) value / maxValue * graphHeight);
                }

                int x = left + i * space + (space - barWidth) / 2;
                int y = top + graphHeight - barHeight;

                if (value < LOW_STOCK_THRESHOLD) {
                    g2.setColor(LOW_STOCK_RED);
                } else {
                    g2.setColor(GRAPH_BLUE);
                }

                if (barHeight > 0) {
                    g2.fillRoundRect(x, y, barWidth, barHeight, 8, 8);
                } else {
                    g2.fillRoundRect(x, top + graphHeight - 3, barWidth, 3, 4, 4);
                }

                g2.setColor(TEXT_DARK);
                g2.setFont(new Font("Arial", Font.BOLD, 10));

                String valueText = String.valueOf(value);
                int valueWidth = g2.getFontMetrics().stringWidth(valueText);

                g2.drawString(
                        valueText,
                        x + (barWidth - valueWidth) / 2,
                        Math.max(12, y - 5)
                );

                String productName = product.name;
                if (productName == null) productName = "";
                productName = shortenName(productName, 9);

                g2.setFont(new Font("Arial", Font.PLAIN, 9));

                int nameWidth = g2.getFontMetrics().stringWidth(productName);
                int labelX = x + (barWidth - nameWidth) / 2;

                if (labelX < left) labelX = left;
                if (labelX + nameWidth > width - right) {
                    labelX = width - right - nameWidth;
                }

                g2.setColor(TEXT_DARK);
                g2.drawString(
                        productName,
                        labelX,
                        top + graphHeight + 17
                );
            }

            g2.dispose();
        }
    }

    private static class PriceLineChartPanel extends JPanel {

        private List<ProductInfo> products = new ArrayList<>();

        public PriceLineChartPanel() {

            setOpaque(false);
            setCursor(new Cursor(Cursor.HAND_CURSOR));

            addMouseListener(new MouseAdapter() {
                @Override
                public void mouseClicked(MouseEvent e) {
                    showClickedProduct(e.getX(), e.getY());
                }
            });
        }

        public void setData(List<ProductInfo> products) {
            this.products = new ArrayList<>(products);
            repaint();
        }

        private void showClickedProduct(int mouseX, int mouseY) {

            if (products.isEmpty()) return;

            int width = getWidth();
            int height = getHeight();

            int left = 55;
            int right = 25;
            int top = 10;
            int bottom = 45;

            int graphWidth = width - left - right;
            int graphHeight = height - top - bottom;

            double maxPrice = 1.0;
            for (ProductInfo product : products) {
                if (product.price > maxPrice) {
                    maxPrice = product.price;
                }
            }

            int count = products.size();

            for (int i = 0; i < count; i++) {

                ProductInfo product = products.get(i);

                int x;
                if (count == 1) {
                    x = left + graphWidth / 2;
                } else {
                    x = left + (int) ((double) i * graphWidth / (count - 1));
                }

                int y = top + graphHeight
                        - (int) (product.price / maxPrice * graphHeight);

                double distance = Math.sqrt(
                        Math.pow(mouseX - x, 2) + Math.pow(mouseY - y, 2)
                );

                Rectangle labelArea = new Rectangle(
                        x - 40, top + graphHeight, 80, 40
                );

                if (distance <= 15 || labelArea.contains(mouseX, mouseY)) {
                    showProductInformation(product);
                    return;
                }
            }
        }

        @Override
        protected void paintComponent(Graphics g) {

            super.paintComponent(g);
            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            int width = getWidth();
            int height = getHeight();

            int left = 55;
            int right = 25;
            int top = 10;
            int bottom = 45;

            int graphWidth = width - left - right;
            int graphHeight = height - top - bottom;

            if (products.isEmpty()) {
                drawNoData(g2, width, height);
                g2.dispose();
                return;
            }

            double maxPrice = 1.0;
            for (ProductInfo product : products) {
                if (product.price > maxPrice) {
                    maxPrice = product.price;
                }
            }

            g2.setFont(new Font("Arial", Font.PLAIN, 10));

            for (int i = 0; i <= 5; i++) {

                int y = top + graphHeight - i * graphHeight / 5;
                double value = i * maxPrice / 5.0;

                g2.setColor(GRID_COLOR);
                g2.drawLine(left, y, width - right, y);

                g2.setColor(TEXT_LIGHT);
                g2.drawString(String.format("R%.0f", value), 8, y + 4);
            }

            g2.setColor(AXIS_COLOR);
            g2.drawLine(left, top, left, top + graphHeight);
            g2.drawLine(left, top + graphHeight, width - right, top + graphHeight);

            int count = products.size();
            double spacing;
            if (count == 1) {
                spacing = 0;
            } else {
                spacing = (double) graphWidth / (count - 1);
            }

            int previousX = 0;
            int previousY = 0;
            boolean previousWasLow = false;

            for (int i = 0; i < count; i++) {

                ProductInfo product = products.get(i);
                double price = product.price;
                boolean isLow = product.quantity < LOW_STOCK_THRESHOLD;

                int x;
                if (count == 1) {
                    x = left + graphWidth / 2;
                } else {
                    x = left + (int) (i * spacing);
                }

                int y = top + graphHeight
                        - (int) (price / maxPrice * graphHeight);

                if (i > 0) {
                    if (isLow || previousWasLow) {
                        g2.setColor(LOW_STOCK_RED);
                    } else {
                        g2.setColor(GRAPH_BLUE);
                    }

                    g2.setStroke(new BasicStroke(
                            3f,
                            BasicStroke.CAP_ROUND,
                            BasicStroke.JOIN_ROUND
                    ));

                    g2.drawLine(previousX, previousY, x, y);
                }

                if (isLow) {
                    g2.setColor(LOW_STOCK_RED);
                } else {
                    g2.setColor(GRAPH_BLUE);
                }

                g2.fillOval(x - 7, y - 7, 14, 14);

                g2.setColor(WHITE);
                g2.fillOval(x - 3, y - 3, 6, 6);

                g2.setColor(TEXT_DARK);
                g2.setFont(new Font("Arial", Font.BOLD, 10));

                String priceText = String.format("R%.2f", price);
                int priceWidth = g2.getFontMetrics().stringWidth(priceText);
                int priceX = x - priceWidth / 2;

                if (priceX < left) priceX = left;
                if (priceX + priceWidth > width - right) {
                    priceX = width - right - priceWidth;
                }

                int priceY = Math.max(12, y - 10);
                g2.drawString(priceText, priceX, priceY);

                String productName = product.name;
                if (productName == null) productName = "";
                productName = shortenName(productName, 9);

                g2.setFont(new Font("Arial", Font.PLAIN, 9));

                int nameWidth = g2.getFontMetrics().stringWidth(productName);
                int nameX = x - nameWidth / 2;

                if (nameX < left) nameX = left;
                if (nameX + nameWidth > width - right) {
                    nameX = width - right - nameWidth;
                }

                g2.drawString(
                        productName,
                        nameX,
                        top + graphHeight + 17
                );

                previousX = x;
                previousY = y;
                previousWasLow = isLow;
            }

            g2.dispose();
        }
    }

    private static class ProductInfo {

        int id;
        String name;
        String category;
        int quantity;
        double price;

        ProductInfo(int id, String name, String category,
                    int quantity, double price) {
            this.id = id;
            this.name = name;
            this.category = category;
            this.quantity = quantity;
            this.price = price;
        }
    }

    private static String shortenName(String name, int maxLength) {

        if (name == null) return "";
        if (name.length() <= maxLength) return name;
        return name.substring(0, maxLength) + "...";
    }

    private static void drawNoData(Graphics2D g2, int width, int height) {

        g2.setColor(TEXT_LIGHT);
        g2.setFont(new Font("Arial", Font.PLAIN, 14));

        String message = "No product data available";
        int textWidth = g2.getFontMetrics().stringWidth(message);

        g2.drawString(
                message,
                (width - textWidth) / 2,
                height / 2
        );
    }

    private static class RoundedButton extends JButton {

        public RoundedButton(String text) {

            super(text);
            setOpaque(false);
            setContentAreaFilled(false);
            setBorderPainted(false);
            setFocusPainted(false);
            setCursor(new Cursor(Cursor.HAND_CURSOR));
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 = (Graphics2D) g.create();
            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            g2.setColor(getBackground());
            g2.fill(new RoundRectangle2D.Double(
                    0, 0, getWidth(), getHeight(), 18, 18
            ));

            g2.setFont(getFont());
            g2.setColor(getForeground());

            FontMetrics fm = g2.getFontMetrics();
            int x = (getWidth() - fm.stringWidth(getText())) / 2;
            int y = (getHeight() - fm.getHeight()) / 2
                    + fm.getAscent();

            g2.drawString(getText(), x, y);
            g2.dispose();
        }

        @Override
        protected void paintBorder(Graphics g) {
            // No border
        }
    }

    private void styleBackButton(JButton button) {

        button.setFont(new Font("Arial", Font.BOLD, 12));
        button.setBackground(new Color(70, 95, 120));
        button.setForeground(Color.WHITE);
        button.setFocusPainted(false);
        button.setBorderPainted(false);
        button.setContentAreaFilled(false);
        button.setOpaque(false);
        button.setCursor(new Cursor(Cursor.HAND_CURSOR));
    }
