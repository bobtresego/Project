    private double getProductPrice(int productID) {

        String sql = "SELECT Price FROM Products WHERE ProductID = ?";

        try (
                Connection con = DBConnection.getConnection();
                PreparedStatement pst = con.prepareStatement(sql)
        ) {
            pst.setInt(1, productID);

            try (ResultSet rs = pst.executeQuery()) {
                if (rs.next()) {
                    return rs.getDouble("Price");
                }
            }

        } catch (Exception e) {
            System.out.println("Price error: " + e.getMessage());
        }

        return 0;
    }

    private void calculateTotal() {

        try {
            String productText = productIDField.getText().trim();
            String quantityText = quantityField.getText().trim();

            if (productText.isEmpty() || quantityText.isEmpty()) {
                priceLabel.setText("R0.00");
                totalLabel.setText("R0.00");
                return;
            }

            int productID = Integer.parseInt(productText);
            int quantity = Integer.parseInt(quantityText);

            if (quantity <= 0) {
                priceLabel.setText("R0.00");
                totalLabel.setText("R0.00");
                return;
            }

            double price = getProductPrice(productID);

            if (price <= 0) {
                priceLabel.setText("Product not found");
                totalLabel.setText("R0.00");
                return;
            }

            double total = price * quantity;

            priceLabel.setText(String.format("R%.2f", price));
            totalLabel.setText(String.format("R%.2f", total));

        } catch (NumberFormatException e) {
            priceLabel.setText("R0.00");
            totalLabel.setText("R0.00");
        }
    }

    private void makeSale() {

        if (productIDField.getText().trim().isEmpty()) {
            JOptionPane.showMessageDialog(
                    this,
                    "Please enter a Product ID.",
                    "Missing Product ID",
                    JOptionPane.WARNING_MESSAGE
            );
            productIDField.requestFocus();
            return;
        }

        if (quantityField.getText().trim().isEmpty()) {
            JOptionPane.showMessageDialog(
                    this,
                    "Please enter the quantity sold.",
                    "Missing Quantity",
                    JOptionPane.WARNING_MESSAGE
            );
            quantityField.requestFocus();
            return;
        }

        int productID;
        int quantity;

        try {
            productID = Integer.parseInt(productIDField.getText().trim());
            quantity = Integer.parseInt(quantityField.getText().trim());
        } catch (NumberFormatException e) {
            JOptionPane.showMessageDialog(
                    this,
                    "Product ID and Quantity must be numbers.",
                    "Invalid Input",
                    JOptionPane.ERROR_MESSAGE
            );
            return;
        }

        if (productID <= 0) {
            JOptionPane.showMessageDialog(
                    this,
                    "Product ID must be greater than 0.",
                    "Invalid Product ID",
                    JOptionPane.WARNING_MESSAGE
            );
            return;
        }

        if (quantity <= 0) {
            JOptionPane.showMessageDialog(
                    this,
                    "Quantity sold must be greater than 0.",
                    "Invalid Quantity",
                    JOptionPane.WARNING_MESSAGE
            );
            return;
        }

        Connection con = null;

        try {
            con = DBConnection.getConnection();

            if (con == null) {
                JOptionPane.showMessageDialog(
                        this,
                        "Could not connect to the database.",
                        "Database Error",
                        JOptionPane.ERROR_MESSAGE
                );
                return;
            }

            con.setAutoCommit(false);

            String productSQL =
                    "SELECT ProductName, Price, Quantity FROM Products WHERE ProductID = ? FOR UPDATE";

            PreparedStatement productStatement = con.prepareStatement(productSQL);
            productStatement.setInt(1, productID);

            ResultSet productResult = productStatement.executeQuery();

            if (!productResult.next()) {
                con.rollback();
                JOptionPane.showMessageDialog(
                        this,
                        "Product ID " + productID + " does not exist.",
                        "Product Not Found",
                        JOptionPane.WARNING_MESSAGE
                );
                productResult.close();
                productStatement.close();
                return;
            }

            String productName = productResult.getString("ProductName");
            double price = productResult.getDouble("Price");
            int currentStock = productResult.getInt("Quantity");

            productResult.close();
            productStatement.close();

            if (quantity > currentStock) {
                con.rollback();
                JOptionPane.showMessageDialog(
                        this,
                        "Not enough stock.\n\n"
                                + "Product: " + productName + "\n"
                                + "Available Stock: " + currentStock + "\n"
                                + "Requested: " + quantity,
                        "Insufficient Stock",
                        JOptionPane.WARNING_MESSAGE
                );
                return;
            }

            double totalPrice = price * quantity;

            String saleSQL =
                    "INSERT INTO Sales (ProductID, QuantitySold, TotalPrice) VALUES (?, ?, ?)";

            PreparedStatement saleStatement = con.prepareStatement(
                    saleSQL,
                    Statement.RETURN_GENERATED_KEYS
            );

            saleStatement.setInt(1, productID);
            saleStatement.setInt(2, quantity);
            saleStatement.setDouble(3, totalPrice);

            int rowsInserted = saleStatement.executeUpdate();

            int saleID = 0;

            if (rowsInserted > 0) {
                ResultSet generatedKeys = saleStatement.getGeneratedKeys();
                if (generatedKeys.next()) {
                    saleID = generatedKeys.getInt(1);
                }
                generatedKeys.close();
            }

            saleStatement.close();

            String updateStockSQL =
                    "UPDATE Products SET Quantity = Quantity - ? WHERE ProductID = ?";

            PreparedStatement updateStatement = con.prepareStatement(updateStockSQL);
            updateStatement.setInt(1, quantity);
            updateStatement.setInt(2, productID);

            int updated = updateStatement.executeUpdate();

            updateStatement.close();

            if (updated != 1) {
                con.rollback();
                JOptionPane.showMessageDialog(
                        this,
                        "Sale could not update product stock.",
                        "Sale Error",
                        JOptionPane.ERROR_MESSAGE
                );
                return;
            }

            con.commit();

            JOptionPane.showMessageDialog(
                    this,
                    "Sale Completed Successfully!\n\n"
                            + "Sale ID: " + saleID + "\n"
                            + "Product: " + productName + "\n"
                            + "Product ID: " + productID + "\n"
                            + "Quantity Sold: " + quantity + "\n"
                            + "Total Price: " + String.format("R%.2f", totalPrice),
                    "Sale Successful",
                    JOptionPane.INFORMATION_MESSAGE
            );

            productIDField.setText("");
            quantityField.setText("");
            priceLabel.setText("R0.00");
            totalLabel.setText("R0.00");

            loadSales();

        } catch (Exception e) {
            if (con != null) {
                try {
                    con.rollback();
                } catch (Exception ignored) {
                }
            }

            JOptionPane.showMessageDialog(
                    this,
                    "Sale could not be completed.\n\nError: " + e.getMessage(),
                    "Sale Error",
                    JOptionPane.ERROR_MESSAGE
            );

            e.printStackTrace();

        } finally {
            if (con != null) {
                try {
                    con.setAutoCommit(true);
                    con.close();
                } catch (Exception ignored) {
                }
            }
        }
    }

    private void loadSales() {

        if (tableModel == null) {
            return;
        }

        String sql =
                "SELECT SaleID, ProductID, QuantitySold, TotalPrice, SaleDate FROM Sales ORDER BY SaleID DESC";

        try (
                Connection con = DBConnection.getConnection();
                PreparedStatement pst = con.prepareStatement(sql);
                ResultSet rs = pst.executeQuery()
        ) {
            tableModel.setRowCount(0);

            while (rs.next()) {
                int saleID = rs.getInt("SaleID");
                int productID = rs.getInt("ProductID");
                int quantitySold = rs.getInt("QuantitySold");
                double totalPrice = rs.getDouble("TotalPrice");
                Timestamp saleDate = rs.getTimestamp("SaleDate");

                tableModel.addRow(new Object[]{
                        saleID,
                        productID,
                        quantitySold,
                        String.format("R%.2f", totalPrice),
                        saleDate
                });
            }

        } catch (Exception e) {
            System.out.println("Unable to load sales: " + e.getMessage());
        }
    }

    private static class RoundedPanel extends JPanel {

        private final int radius;

        RoundedPanel(int radius) {
            this.radius = radius;
            setOpaque(false);
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 = (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            g2.setColor(getBackground());
            g2.fillRoundRect(0, 0, getWidth(), getHeight(), radius, radius);

            g2.dispose();
            super.paintComponent(g);
        }
    }

    private static class RoundedButton extends JButton {

        RoundedButton(String text) {

            super(text);

            setFocusPainted(false);
            setBorderPainted(false);
            setContentAreaFilled(false);
            setOpaque(false);
            setCursor(new Cursor(Cursor.HAND_CURSOR));
            setHorizontalAlignment(SwingConstants.CENTER);
            setVerticalAlignment(SwingConstants.CENTER);
            setMargin(new Insets(0, 0, 0, 0));
        }

        @Override
        protected void paintComponent(Graphics g) {

            Graphics2D g2 = (Graphics2D) g.create();

            g2.setRenderingHint(
                    RenderingHints.KEY_ANTIALIASING,
                    RenderingHints.VALUE_ANTIALIAS_ON
            );

            Color buttonColor = getBackground();

            if (getModel().isPressed()) {
                buttonColor = buttonColor.darker();
            } else if (getModel().isRollover()) {
                buttonColor = buttonColor.brighter();
            }

            g2.setColor(buttonColor);

            g2.fillRoundRect(0, 0, getWidth(), getHeight(), 12, 12);

            g2.dispose();
            super.paintComponent(g);
        }
    }
